# Baseline profile for Yaadein: tells Android which code to compile ahead of time when the app is installed, so the
# first launches do not run in the slow interpreter. The libraries (Compose, Navigation, Lifecycle, Room, Media3...)
# ship their own profiles, which the build merges in; these rules add Yaadein's own code, which no library covers.
# Wildcards are expanded by the build against the final (R8-renamed) classes.
#
# H = hot, S = used during startup, P = used after startup, L...; = class loaded at startup.
#
# Startup path: process start, settings + vault check, lock screen or welcome screen, Home and the timeline.
HSPLcom/yaadein/app/**->**(**)**
HSPLcom/yaadein/core/ui/**->**(**)**
HSPLcom/yaadein/data/**->**(**)**
HSPLcom/yaadein/domain/**->**(**)**
HSPLcom/yaadein/feature/timeline/**->**(**)**
HSPLcom/yaadein/feature/lock/**->**(**)**
HSPLcom/yaadein/feature/onboarding/**->**(**)**
Lcom/yaadein/app/**;
Lcom/yaadein/core/ui/**;
Lcom/yaadein/data/**;
Lcom/yaadein/domain/**;
Lcom/yaadein/feature/timeline/**;
Lcom/yaadein/feature/lock/**;
Lcom/yaadein/feature/onboarding/**;
# Everything else the user reaches after startup (People, Relive, capture, memory, reels, settings): compiled too, so
# opening a tab or a screen for the first time is not interpreted either.
HPLcom/yaadein/feature/**->**(**)**
